% =====================================================================
% Reusable two-column designer CV template (XeTeX / Tectonic)
% Layout, colors, and macros only. Content lives in the instance file:
%   \input{../templates/cv-template.tex}
%   \begin{document} ... \end{document}
%
% Overridable before \begin{document}:
%   \renewcommand{\cvname}{...}   name shown in the footer
%   \renewcommand{\cvpages}{2}    page count shown in the footer
%   place a photo at cv/assets/photo.jpg to replace the placeholder
%
% Macros provided:
%   \photobox                          photo or placeholder
%   \prof{0..5}                        proficiency dots
%   \mainsection{TITLE} \sidesection{TITLE}
%   \cvevent{dates}{title}{subtitle}{bullets}
%   \patent{id}{title}{url}
%   \lang{name}{note}{0..5}  \tool{name}{note}{0..5}
%   \toolgroup{group}{tools}           tool group without proficiency dots
%   cvitems                            bulleted list environment
%   cvhl                               checkmark list for highlights
% =====================================================================
\documentclass[10pt,a4paper]{article}

% ---------- Fonts / engine (XeTeX via Tectonic) ----------
\usepackage{fontspec}
\usepackage[default,scale=1.0]{sourcesanspro}
\usepackage{textcomp}

% ---------- Layout ----------
\usepackage[a4paper,left=8mm,right=13mm,top=12mm,bottom=9mm]{geometry}
\usepackage{xcolor}
\usepackage{tikz}
\usepackage{eso-pic}
\usepackage{enumitem}
\usepackage{graphicx}
\usepackage{ragged2e}
\usepackage{fontawesome5}
\usepackage{fancyhdr}
\usepackage{hyperref}

% ---------- Overridable parameters ----------
\newcommand{\cvname}{David Matzek}
\newcommand{\cvpages}{2}
\newcommand{\cvphoto}{../assets/photo.jpg}

% ---------- Colors ----------
\definecolor{accent}{HTML}{1F9ED4}
\definecolor{sidebarbg}{HTML}{EDEDED}
\definecolor{dotempty}{HTML}{CBCBCB}
\definecolor{subgray}{HTML}{8A8A8A}
\definecolor{maintext}{HTML}{262626}
\color{maintext}
\hypersetup{colorlinks=true,urlcolor=accent,linkcolor=accent}

% ---------- Dimensions ----------
\newlength{\barwidth}\setlength{\barwidth}{76mm}
\newlength{\sidebarw}\setlength{\sidebarw}{57mm}
\newlength{\mainw}\setlength{\mainw}{112mm}
\newlength{\cvgap}\setlength{\cvgap}{19mm}
\setlength{\parindent}{0pt}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\fancyfoot[R]{\footnotesize\color{subgray}\cvname\ \textbullet\ Page \thepage\,/\,\cvpages}

% ---------- Full-bleed sidebar background on every page ----------
\AddToShipoutPictureBG{%
  \AtPageLowerLeft{\color{sidebarbg}\rule{\barwidth}{\paperheight}}%
}

% ---------- Proficiency dots (filled out of 5) ----------
\newcommand{\dotone}[1]{\tikz[baseline]{\fill[#1] (0,0.5ex) circle (0.5ex);}\hspace{0.35ex}}
\newcommand{\prof}[1]{%
  \ifcase#1
    \dotone{dotempty}\dotone{dotempty}\dotone{dotempty}\dotone{dotempty}\dotone{dotempty}%
  \or\dotone{accent}\dotone{dotempty}\dotone{dotempty}\dotone{dotempty}\dotone{dotempty}%
  \or\dotone{accent}\dotone{accent}\dotone{dotempty}\dotone{dotempty}\dotone{dotempty}%
  \or\dotone{accent}\dotone{accent}\dotone{accent}\dotone{dotempty}\dotone{dotempty}%
  \or\dotone{accent}\dotone{accent}\dotone{accent}\dotone{accent}\dotone{dotempty}%
  \or\dotone{accent}\dotone{accent}\dotone{accent}\dotone{accent}\dotone{accent}%
  \fi}

% ---------- Section headers ----------
\newcommand{\mainsection}[1]{\par\vspace{3.6mm}{\color{accent}\fontsize{15}{17}\selectfont #1}\par\vspace{1mm}{\color{accent}\rule{\linewidth}{1.1pt}}\par\vspace{2.4mm}}
\newcommand{\sidesection}[1]{\par\vspace{3.4mm}{\color{accent}\fontsize{12.5}{14}\selectfont #1}\par\vspace{0.8mm}{\color{accent}\rule{\linewidth}{0.5pt}}\par\vspace{1.6mm}}

% ---------- Bulleted lists ----------
\newlist{cvitems}{itemize}{1}
\setlist[cvitems]{label={\color{accent}\raisebox{0.35ex}{\fontsize{4}{4}\selectfont\faSquare}},leftmargin=3.6mm,labelsep=1.6mm,topsep=0.3mm,itemsep=0.5mm,parsep=0pt,before=\RaggedRight}

\newlist{cvhl}{itemize}{1}
\setlist[cvhl]{label={\color{accent}\raisebox{0.1ex}{\footnotesize\faCheck}},leftmargin=5mm,labelsep=1.6mm,topsep=0.3mm,itemsep=1.4mm,parsep=0pt}

% ---------- Timeline event: dates | title / subtitle / bullets ----------
\newcommand{\cvevent}[4]{%
  \begin{minipage}[t]{16mm}\raggedright\footnotesize\color{subgray}#1\end{minipage}\hspace{2.5mm}%
  \begin{minipage}[t]{\dimexpr\linewidth-18.5mm\relax}%
    {\bfseries #2}\\[0.2mm]{\color{subgray}\footnotesize #3}%
    \vspace{0.6mm}#4%
  \end{minipage}\par\vspace{2.4mm}%
}

% ---------- Patent row: id | title ----------
\newcommand{\patent}[3]{%
  \begin{minipage}[t]{32mm}\raggedright\small\color{accent}\href{#3}{#1}\end{minipage}\hspace{2.5mm}%
  \begin{minipage}[t]{\dimexpr\linewidth-34.5mm\relax}\small #2\end{minipage}\par\vspace{1.8mm}%
}

% ---------- Sidebar language / tool blocks ----------
\newcommand{\lang}[3]{{\bfseries #1}\\[0.1mm]{\footnotesize\color{subgray}#2}\\[0.8mm]\prof{#3}\par\vspace{1.8mm}}
\newcommand{\tool}[3]{{\color{accent}\bfseries #1}\\[0.1mm]{\footnotesize\color{subgray}#2}\\[0.8mm]\prof{#3}\par\vspace{1.8mm}}
\newcommand{\toolgroup}[2]{{\color{accent}\bfseries #1}\\[0.1mm]{\footnotesize\color{subgray}#2}\par\vspace{2.4mm}}

% ---------- Photo (drop cv/assets/photo.jpg to replace placeholder) ----------
\newcommand{\photobox}{%
  \IfFileExists{\cvphoto}{\includegraphics[width=\linewidth]{\cvphoto}}{%
    \begin{tikzpicture}\fill[gray!40] (0,0) rectangle (\linewidth,42mm);\node at (0.5\linewidth,21mm){\color{white}\fontsize{40}{40}\selectfont\faUser};\end{tikzpicture}}%
}
